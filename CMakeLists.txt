cmake_minimum_required(VERSION 3.16)
project(gchd VERSION 0.3.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
option(GCHD_WERROR "Treat compiler warnings as errors" OFF)
option(GCHD_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(gchd_options INTERFACE)
target_compile_features(gchd_options INTERFACE cxx_std_11)
target_compile_options(gchd_options INTERFACE -Wall -Wextra)
if(GCHD_WERROR)
  target_compile_options(gchd_options INTERFACE -Werror)
endif()
if(GCHD_SANITIZE)
  target_compile_options(gchd_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(gchd_options INTERFACE -fsanitize=address,undefined)
endif()
add_subdirectory(src)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
set(GCHD_UDEV_RULES_DIR "lib/udev/rules.d" CACHE STRING "udev rule install directory (relative to prefix or absolute)")
install(FILES udev-rules/55-elgato-game-capture.rules DESTINATION ${GCHD_UDEV_RULES_DIR})
install(FILES README.md CREDITS.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR})

install(PROGRAMS scripts/gchd-record.sh DESTINATION ${CMAKE_INSTALL_BINDIR})
